★ Side B · All three, together

Read health.js:

start(game) { game.health = 3; }          // a variable

onZombieHit(game) {
  game.health = game.health - 1;         // change it
  if (game.health <= 0) game.lose("Got you!");  // an if
}

★ Side B · Build it together

Now fight back — same three ideas.

start(game) {
  game.zombies.forEach(z => z.hp = 2);   // a variable
}
use(player, game) {
  const z = game.zombieAt(game.squareInFront(player));
  if (z) {
    z.hp = z.hp - 1;                     // change it
    if (z.hp <= 0) game.removeZombieAt(z);  // an if
  }
}
